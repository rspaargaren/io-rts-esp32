**Background:**
- `create_1w_hmac`: builds a 16-byte IV from frame data + seq, then AES-128-ECB-encrypts it with `key_1w` → take first 6 bytes as HMAC. Uses existing `aes128_encrypt`.
- `encrypt_1w_key`: builds a node-based IV (node_id repeated across 16 bytes), then AES-128-CFB128-encrypts `key_in` with the protocol-defined TRANSFER_KEY (already in `iohome_constants.h`). Uses mbedtls directly.

- [ ] **Step 2: Implement create_1w_hmac in iohome_crypto.cpp**

Add at the end of `iohome_crypto.cpp`, inside `namespace iohome { namespace crypto {`:

```cpp
    bool create_1w_hmac(
        const uint8_t *frame_cmd_data, size_t frame_len,
        const uint8_t seq[2],
        const uint8_t key[AES_KEY_SIZE],
        uint8_t hmac_out[HMAC_SIZE])
    {
        // Build 16-byte IV: frame bytes [0..7] in positions [0..7], custom
        // checksum in [8..9], seq in [10..11], 0x55 padding in [12..15].
        uint8_t iv[AES_BLOCK_SIZE] = {};
        uint8_t chk1 = 0, chk2 = 0;
        for (size_t i = 0; i < frame_len; i++) {
            compute_checksum(frame_cmd_data[i], chk1, chk2);
            if (i < 8) iv[i] = frame_cmd_data[i];
        }
        for (size_t j = frame_len; j < 8; j++) iv[j] = 0x55;
        iv[8]  = chk1;
        iv[9]  = chk2;
        iv[10] = seq[0];
        iv[11] = seq[1];
        iv[12] = iv[13] = iv[14] = iv[15] = 0x55;

        uint8_t block[AES_BLOCK_SIZE];
        if (!aes128_encrypt(iv, key, block)) return false;
        memcpy(hmac_out, block, HMAC_SIZE);
        return true;
    }
```

- [ ] **Step 3: Implement encrypt_1w_key in iohome_crypto.cpp**

Add immediately after `create_1w_hmac`, still inside the namespace:

```cpp
    bool encrypt_1w_key(
        const uint8_t controller_node_id[NODE_ID_SIZE],
        const uint8_t key_in[AES_KEY_SIZE],
        uint8_t enc_key_out[AES_KEY_SIZE])
    {
        // IV = controller node repeated: positions 0-2, 3-5, 6-8, 9-11, 12-14 = node[0..2], pos 15 = node[0]
        // controller_node_id is mOwnNodeId (our A1B1C3), NOT the paired device's ID.
        uint8_t iv[AES_BLOCK_SIZE] = {};
        for (int i = 0; i < 13; i += 3) {
            iv[i]   = controller_node_id[0];
            iv[i+1] = controller_node_id[1];
            iv[i+2] = controller_node_id[2];
        }
        iv[15] = controller_node_id[0];

        // AES-128-CFB128 encrypt key_in with TRANSFER_KEY using the node IV
        mbedtls_aes_context aes_ctx;
        mbedtls_aes_init(&aes_ctx);
        int ret = mbedtls_aes_setkey_enc(&aes_ctx, TRANSFER_KEY, 128);
        if (ret != 0) { mbedtls_aes_free(&aes_ctx); return false; }
        size_t offset = 0;
        ret = mbedtls_aes_crypt_cfb128(&aes_ctx, MBEDTLS_AES_ENCRYPT,
                                        AES_KEY_SIZE, &offset, iv,
                                        key_in, enc_key_out);
        mbedtls_aes_free(&aes_ctx);
        return ret == 0;
    }
```

## Task 3 — Io1WControl Class

**Files:**

**Interfaces:**
- Consumes: `iohome::IoHomeControl::TransmitFrame`, `iohome::crypto::create_1w_hmac`, `iohome::crypto::encrypt_1w_key`, `iohome::init_frame`, `iohome::set_destination`, `iohome::set_source`, `iohome::set_command`
- Produces:
  - `Io1WControl(IoHomeControl*, const std::string& own_node_id)`
  - `bool PairDevice(iohome::IoDeviceInformation& info)` — generates key, sends PAIR+ADD, sets info fields
  - `bool UnpairDevice(iohome::IoDeviceInformation& info)` — sends REMOVE
  - `bool Send(iohome::IoDeviceInformation& info, float position_pct)` — sends CMD 0x00, increments seq

**Protocol notes:**
- Broadcast target: `(uint8_t(device_type) << 6) | 0x3F` → stored in dest_node[1..2], dest_node[0]=0x00
- CMD 0x00 payload (13 bytes): origin(1=user) + acei(0x43) + main[2] + fp1(0x00) + seq[2] + hmac[6]
- main for position: `uint16_t m = uint16_t(position_pct * 512.0f); main[0]=m>>8; main[1]=m&0xFF`
- Open (0%): main=0x0000. Close (100%): main=0xC800. Position X%: main=X*512.
- CMD 0x2E payload (9 bytes): data(0x00) + seq[2] + hmac[6]. HMAC frame_data = {0x2E, 0x00}
- CMD 0x30 payload (20 bytes): enc_key[16] + man_id(1) + data(0x01) + seq[2]. No HMAC.
- CMD 0x39 payload: same as 0x2E. HMAC frame_data = {0x39, 0x00}
- Frames use `init_frame(f, false /*1W*/, true /*start*/, true /*end*/, info.is_low_power)`
- TransmitFrame called 4× per command on FREQUENCY_CHANNEL_2 with LONG_PREAMBLE_LENGTH

#include "iohome_frame.hpp"

#include "iohome_constants.h"

static const char *TAG = "Io1WControl";

namespace iohome {

Io1WControl::Io1WControl(IoHomeControl *io_home, const std::string &own_node_id)
    : mIoHome(io_home)
{
    memset(mOwnNodeId, 0, NODE_ID_SIZE);
    // Convert hex string "A1B1C3" → bytes {0xA1, 0xB1, 0xC3}
    for (size_t i = 0; i < NODE_ID_SIZE && i * 2 + 1 < own_node_id.size(); i++) {
        char b[3] = {own_node_id[i*2], own_node_id[i*2+1], 0};
        mOwnNodeId[i] = (uint8_t)strtoul(b, nullptr, 16);
    }
}

void Io1WControl::BuildBroadcastTarget(uint8_t dest[NODE_ID_SIZE], DeviceType type) const
{
    uint16_t bcast = (static_cast<uint16_t>(type) << 6) | 0x3F;
    dest[0] = 0x00;
    dest[1] = (uint8_t)(bcast >> 8);
    dest[2] = (uint8_t)(bcast & 0xFF);
}

void Io1WControl::TransmitFrame4x(const IoFrame &frame) const
{
    for (int i = 0; i < 4; i++)
        mIoHome->TransmitFrame(frame, FREQUENCY_CHANNEL_2, LONG_PREAMBLE_LENGTH);
}

bool Io1WControl::PairDevice(IoDeviceInformation &info)
{
    // Generate a random 16-byte AES key
    esp_fill_random(info.key_1w, AES_KEY_SIZE);
    info.sequence_1w   = (uint16_t)(esp_random() & 0xFFFF);
    if (info.sequence_1w == 0) info.sequence_1w = 1; // never start at 0
    info.protocol_mode = ProtocolMode::PROTO_1W;

    uint8_t dest[NODE_ID_SIZE];
    BuildBroadcastTarget(dest, info.device_type);

    // --- CMD 0x2E (PAIR) ---
    {
        uint8_t seq[2] = {(uint8_t)(info.sequence_1w >> 8), (uint8_t)(info.sequence_1w & 0xFF)};
        info.sequence_1w++;

        // payload: data(0x00) | seq[2] | hmac[6]
        uint8_t frame_for_hmac[2] = {0x2E, 0x00};
        uint8_t hmac[HMAC_SIZE];
        if (!crypto::create_1w_hmac(frame_for_hmac, sizeof(frame_for_hmac), seq, info.key_1w, hmac)) {
            ESP_LOGE(TAG, "PairDevice: HMAC failed");
            return false;
        }

        uint8_t params[9];
        params[0] = 0x00; // data
        params[1] = seq[0]; params[2] = seq[1];
        memcpy(&params[3], hmac, HMAC_SIZE);

        IoFrame frame;
        init_frame(frame, false /*1W*/, true, true, info.is_low_power);
        set_destination(frame, dest);
        set_source(frame, mOwnNodeId);
        set_command(frame, 0x2E, params, sizeof(params));
        TransmitFrame4x(frame);
        ESP_LOGI(TAG, "PairDevice: PAIR (0x2E) sent");
    }

    // --- CMD 0x30 (ADD) ---
    {
        uint8_t seq[2] = {(uint8_t)(info.sequence_1w >> 8), (uint8_t)(info.sequence_1w & 0xFF)};
        info.sequence_1w++;

        uint8_t enc_key[AES_KEY_SIZE];
        // encrypt_1w_key uses OUR node ID (controller), not the device's.
        // The device decrypts using the src_node field in the frame header.
        if (!crypto::encrypt_1w_key(mOwnNodeId, info.key_1w, enc_key)) {
            ESP_LOGE(TAG, "PairDevice: key encrypt failed");
            return false;
        }

        // payload: enc_key[16] | man_id(1) | data(0x01) | seq[2]
        uint8_t params[20];
        memcpy(params, enc_key, AES_KEY_SIZE);
        params[16] = static_cast<uint8_t>(info.manufacturer);
        params[17] = 0x01; // data
        params[18] = seq[0]; params[19] = seq[1];

        IoFrame frame;
        init_frame(frame, false /*1W*/, true, true, info.is_low_power);
        set_destination(frame, dest);
        set_source(frame, mOwnNodeId);
        set_command(frame, 0x30, params, sizeof(params));
        TransmitFrame4x(frame);
        ESP_LOGI(TAG, "PairDevice: ADD (0x30) sent");
    }

    return true;
}

bool Io1WControl::Send(IoDeviceInformation &info, float position_pct)
{
    uint8_t dest[NODE_ID_SIZE];
    BuildBroadcastTarget(dest, info.device_type);

    uint8_t seq[2] = {(uint8_t)(info.sequence_1w >> 8), (uint8_t)(info.sequence_1w & 0xFF)};
    info.sequence_1w++;

    // main = position_pct * 512 (0x0000=open, 0xC800=closed)
    uint16_t main_param = (uint16_t)roundf(position_pct * 512.0f);

    // payload: origin(1) | acei(1) | main[2] | fp1(1) | seq[2] | hmac[6]  = 13 bytes
    // HMAC covers: {0x00, origin, acei, main[0], main[1], fp1} (cmd byte + first 5 data bytes)
    uint8_t origin = 0x01;  // user-initiated
    uint8_t acei   = 0x43;

    uint8_t frame_for_hmac[6] = {
        0x00,                          // command byte
        origin, acei,
        (uint8_t)(main_param >> 8), (uint8_t)(main_param & 0xFF),
        0x00                           // fp1
    };
    uint8_t hmac[HMAC_SIZE];
    if (!crypto::create_1w_hmac(frame_for_hmac, sizeof(frame_for_hmac), seq, info.key_1w, hmac)) {
        ESP_LOGE(TAG, "Send: HMAC failed");
        return false;
    }

    uint8_t params[13];
    params[0] = origin;
    params[1] = acei;
    params[2] = (uint8_t)(main_param >> 8);
    params[3] = (uint8_t)(main_param & 0xFF);
    params[4] = 0x00; // fp1
    params[5] = seq[0]; params[6] = seq[1];
    memcpy(&params[7], hmac, HMAC_SIZE);

    IoFrame frame;
    init_frame(frame, false /*1W*/, true, true, info.is_low_power);
    set_destination(frame, dest);
    set_source(frame, mOwnNodeId);
    set_command(frame, 0x00, params, sizeof(params));
    TransmitFrame4x(frame);
    return true;
}

} // namespace iohome

    std::string IoRtsManager::Pair1WDevice(const std::string &name,
                                            iohome::DeviceType type,
                                            iohome::Manufacturer manufacturer)
    {
        // Generate a node ID: random 3 bytes, use as device ID
        uint8_t rand_id[3];
        esp_fill_random(rand_id, 3);
        char id_str[7];
        snprintf(id_str, sizeof(id_str), "%02X%02X%02X", rand_id[0], rand_id[1], rand_id[2]);

        iohome::IoDeviceInformation info = {};
        info.protocol_mode = iohome::ProtocolMode::PROTO_1W;
        memcpy(info.node_id, rand_id, iohome::NODE_ID_SIZE);
        snprintf(info.name, sizeof(info.name), "%s", name.c_str());
        info.device_type  = type;
        info.manufacturer = manufacturer;
        info.is_low_power = false;

        if (!mIo1W->PairDevice(info)) {
            ESP_LOGE("IoRtsManager", "Pair1WDevice: radio TX failed");
            return "";
        }

        // Store in memory and on flash
        iohome::IoDevice dev = {};
        dev.info = info;
        dev.position = iohome::UNKNOWN_POSITION;
        dev.target   = iohome::UNKNOWN_POSITION;
        dev.is_stopped = true;
        dev.transit_time_ms = 0;
        {
            std::lock_guard<std::mutex> lock(mIoDevicesMutex);
            mIoDevices[id_str] = dev;
        }
        Helpers::StoredIoDevice sd;
        sd.device = dev;
        Helpers::DeviceStorage::SaveIoDevice(id_str, sd);
        ESP_LOGI("IoRtsManager", "Pair1WDevice: paired %s as %s", name.c_str(), id_str);
        return id_str;
    }

    bool IoRtsManager::Unpair1WDevice(const std::string &deviceId)
    {
        mIoDevicesMutex.lock();
        auto it = mIoDevices.find(deviceId);
        if (it == mIoDevices.end() || it->second.info.protocol_mode != iohome::ProtocolMode::PROTO_1W) {
            mIoDevicesMutex.unlock();
            return false;
        }
        auto info = it->second.info; // copy
        mIoDevicesMutex.unlock();

        mIo1W->UnpairDevice(info);
        // Remove from memory and storage
        {
            std::lock_guard<std::mutex> lock(mIoDevicesMutex);
            mIoDevices.erase(deviceId);
        }
        Helpers::DeviceStorage::RemoveIoDevice(deviceId);
        return true;
    }

// Global for 1W calibration
let _calib1WDeviceId = null;
let _calib1WOpenTime = null;

function open1WPairModal() {
    document.getElementById('1w-step-1').style.display = '';
    document.getElementById('1w-step-2').style.display = 'none';
    document.getElementById('1w-step-3').style.display = 'none';
    document.getElementById('modal1WPair').style.display = 'flex';
    // 30s countdown
    let secs = 30;
    const countdown = document.getElementById('1w-countdown');
    countdown.textContent = secs + 's';
    const timer = setInterval(() => {
        secs--;
        countdown.textContent = secs + 's';
        if (secs <= 0) clearInterval(timer);
    }, 1000);
}

function close1WModal() {
    document.getElementById('modal1WPair').style.display = 'none';
}

function start1WPair() {
    document.getElementById('1w-step-1').style.display = 'none';
    document.getElementById('1w-step-2').style.display = '';
}

async function submit1WPair() {
    const name = document.getElementById('1w-name').value.trim();
    const type = parseInt(document.getElementById('1w-type').value);
    const mfr  = parseInt(document.getElementById('1w-manufacturer').value);
    document.getElementById('1w-step-2').style.display = 'none';
    document.getElementById('1w-step-3').style.display = '';
    const msg = document.getElementById('1w-result-msg');
    msg.textContent = 'Sending pairing frames...';
    try {
        const r = await fetch('/api/action', {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify({action: 'pair1w', name, device_type: type, manufacturer: mfr})
        });
        const d = await r.json();
        if (d.success) {
            msg.textContent = 'Paired successfully! Device ID: ' + d.deviceId;
            refreshDevices();
        } else {
            msg.textContent = 'Pairing failed: ' + (d.error || 'unknown error');
        }
    } catch(e) {
        msg.textContent = 'Network error: ' + e.message;
    }
}

// 1W Manual Calibration
function open1WCalibModal(deviceId) {
    _calib1WDeviceId = deviceId;
    _calib1WOpenTime = null;
    ['1','2','3','4','5'].forEach(n => {
        document.getElementById('1w-calib-step-' + n).style.display = n === '1' ? '' : 'none';
    });
    document.getElementById('modal1WCalib').style.display = 'flex';
}

function close1WCalibModal() {
    document.getElementById('modal1WCalib').style.display = 'none';
}

async function calib1WSendOpen() {
    await fetch('/api/action', {method:'POST', headers:{'Content-Type':'application/json'},
        body: JSON.stringify({action:'open', deviceId: _calib1WDeviceId})});
    document.getElementById('1w-calib-step-1').style.display = 'none';
    document.getElementById('1w-calib-step-2').style.display = '';
}

function calib1WOpenDone() {
    _calib1WOpenTime = Date.now();
    document.getElementById('1w-calib-step-2').style.display = 'none';
    document.getElementById('1w-calib-step-3').style.display = '';
}

async function calib1WSendClose() {
    await fetch('/api/action', {method:'POST', headers:{'Content-Type':'application/json'},
        body: JSON.stringify({action:'close', deviceId: _calib1WDeviceId})});
    document.getElementById('1w-calib-step-3').style.display = 'none';
    document.getElementById('1w-calib-step-4').style.display = '';
}

async function calib1WCloseDone() {
    const elapsed = Date.now() - _calib1WOpenTime;
    document.getElementById('1w-calib-step-4').style.display = 'none';
    document.getElementById('1w-calib-step-5').style.display = '';
    const r = document.getElementById('1w-calib-result');
    r.textContent = 'Saving transit time: ' + elapsed + ' ms...';
    try {
        const resp = await fetch('/api/action', {method:'POST', headers:{'Content-Type':'application/json'},
            body: JSON.stringify({action:'setTransitTime', deviceId: _calib1WDeviceId, transit_time_ms: elapsed})});
        const d = await resp.json();
        r.textContent = d.success
            ? 'Done! Transit time set to ' + Math.round(elapsed/1000) + 's.'
            : 'Failed: ' + (d.error || '');
    } catch(e) { r.textContent = 'Error: ' + e.message; }
}
